## A Algorithmic Details

<div class="mineru-algorithm" style="white-space: pre-wrap; font-family:monospace;">
Algorithm 1 CFM for Conditional Latent Vector Generation
Require: Pre-computed latent vectors and conditions $\mathcal{Z} = \{(\mathbf{z}^{(e)}, e)\}_{e \in E_{seen}}$
Ensure: Trained CFM network $v_{\xi}$
// Stage 1: Training the Conditional Vector Field
procedure TRAINCFM($\mathcal{Z}$)
    Initialize CFM network parameters $\xi$.
    for each training step do
    Sample batch $\{(\mathbf{z}_1^{(i)}, e^{(i)})\}_{i=1}^B$ from $\mathcal{Z}$.
    Sample priors $\mathbf{z}_0^{(i)} \sim \mathcal{N}(0, I)$ and times $t^{(i)} \sim \mathcal{U}(0, 1)$.
    $\mathbf{z}_t^{(i)} \leftarrow (1 - t^{(i)})\mathbf{z}_0^{(i)} + t^{(i)}\mathbf{z}_1^{(i)} \triangleright$ Construct path points
    $\mathbf{u}^{(i)} \leftarrow \mathbf{z}_1^{(i)} - \mathbf{z}_0^{(i)} \triangleright$ Define target velocities
    $\mathcal{L}_{CFM} \leftarrow \frac{1}{B} \sum_{i=1}^B \|v_{\xi}(\mathbf{z}_t^{(i)}, t^{(i)}, e^{(i)}) - \mathbf{u}^{(i)}\|^2 \triangleright$ Loss
    Update $\xi$ by descending the gradient of $\mathcal{L}_{CFM}$.
    end for
end procedure
// Stage 2: Generating Latent Vectors via Inference
procedure GENERATELATENTVECTOR($v_{\xi}, e^{\text{new}}, N$)
    Sample prior $\mathbf{z}_0 \sim \mathcal{N}(0, I)$.
    for $k = 0, \ldots, N-1$ do
    $\mathbf{z}_{k+1} \leftarrow \mathbf{z}_k + \frac{1}{N} \cdot v_{\xi}(\mathbf{z}_k, k/N, e^{\text{new}}) \triangleright$ Forward Euler step
    end for
    return $\mathbf{z}_N$ ▷ Latent vector for new environment
end procedure
